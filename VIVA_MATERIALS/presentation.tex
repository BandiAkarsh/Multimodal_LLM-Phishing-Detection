\begin{frame}{Three-Tier Detection}
    \begin{enumerate}
        \item \textbf{Tier 1: Validation}
        \begin{itemize}
            \item URL canonicalization
            \item SSRF protection
            \item Length/scheme checks
        \end{itemize}
        
        \item \textbf{Tier 2: Static ML}
        \begin{itemize}
            \item 93 features extracted
            \item Random Forest + XGBoost ensemble
            \item 180ms latency (offline)
        \end{itemize}
        
        \item \textbf{Tier 3: Online Content} (optional)
        \begin{itemize}
            \item Web scraping (Playwright)
            \item Toolkit signature detection
            \item MLLM for AI-generated content
        \end{itemize}
    \end{enumerate}
\end{frame}
